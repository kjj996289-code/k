import java.util.*;
class Slip2Q1 {
 static boolean armstrong(int n) {
  int x=n, digits=String.valueOf(n).length(), sum=0;
  while(x>0){int d=x%10; sum+=Math.pow(d,digits); x/=10;}
  return sum==n;
 }
 public static void main(String[] args) {
  Scanner sc=new Scanner(System.in);
  System.out.print("Enter start and end: ");
  int a=sc.nextInt(), b=sc.nextInt();
  for(int i=a;i<=b;i++) if(armstrong(i)) System.out.print(i+" ");
 }
}

import java.util.*;
class Slip4Q1 {
 static void print(int[][] a){for(int[] r:a){for(int x:r)System.out.print(x+" ");System.out.println();}}
 public static void main(String[] args){
  Scanner s=new Scanner(System.in); int ch;
  do{
   System.out.println("1.Add 2.Multiply 3.Transpose 4.Exit"); ch=s.nextInt();
   if(ch==4)break;
   System.out.print("Rows and columns: "); int r=s.nextInt(), c=s.nextInt(); int[][] a=new int[r][c];
   System.out.println("Enter matrix:"); for(int i=0;i<r;i++)for(int j=0;j<c;j++)a[i][j]=s.nextInt();
   if(ch==3){for(int j=0;j<c;j++){for(int i=0;i<r;i++)System.out.print(a[i][j]+" ");System.out.println();}}
   else {
    System.out.print("Second matrix rows and columns: "); int r2=s.nextInt(),c2=s.nextInt();
    int[][] b=new int[r2][c2]; for(int i=0;i<r2;i++)for(int j=0;j<c2;j++)b[i][j]=s.nextInt();
    if(ch==1 && r==r2 && c==c2){int[][] z=new int[r][c];for(int i=0;i<r;i++)for(int j=0;j<c;j++)z[i][j]=a[i][j]+b[i][j];print(z);}
    else if(ch==2 && c==r2){int[][] z=new int[r][c2];for(int i=0;i<r;i++)for(int j=0;j<c2;j++)for(int k=0;k<c;k++)z[i][j]+=a[i][k]*b[k][j];print(z);}
    else System.out.println("Invalid matrix sizes");
   }
  }while(ch!=4);
 }
}
